import { useState, useCallback, useEffect } from 'react'
import { Outlet } from 'react-router-dom'
import Navbar from '../components/Navbar'
import HotkeyCheatsheet from '../components/HotkeyCheatsheet'
import SettingsPanel from '../components/SettingsPanel'
import { useSettings } from '../store/useSettings'

export default function MainLayout() {
  /* в•ђв•ђв•ђ РЎРѕСЃС‚РѕСЏРЅРёСЏ РїР°РЅРµР»РµР№ в•ђв•ђв•ђ */
  const [cheatsheetOpen, setCheatsheetOpen] = useState(false)
  const [settingsOpen, setSettingsOpen] = useState(false)
  const [shuffleActive, setShuffleActive] = useState(false)
  const [autoplayActive, setAutoplayActive] = useState(true)

  /* в•ђв•ђв•ђ РќР°СЃС‚СЂРѕР№РєРё РёР· Zustand в•ђв•ђв•ђ */
  const settings = useSettings()

  /* в•ђв•ђв•ђ РЎРёРЅС…СЂРѕРЅРёР·Р°С†РёСЏ С‚РµРјС‹ в•ђв•ђв•ђ */
  useEffect(() => {
    const html = document.documentElement
    if (settings.theme === 'auto') {
      const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches
      html.dataset.theme = prefersDark ? 'dark' : ''
      const mq = window.matchMedia('(prefers-color-scheme: dark)')
      const handler = (e: MediaQueryListEvent) => {
        html.dataset.theme = e.matches ? 'dark' : ''
      }
      mq.addEventListener('change', handler)
      return () => mq.removeEventListener('change', handler)
    } else {
      html.dataset.theme = settings.theme === 'dark' ? 'dark' : ''
    }
  }, [settings.theme])

  /* в•ђв•ђв•ђ РЎРёРЅС…СЂРѕРЅРёР·Р°С†РёСЏ Р°РєС†РµРЅС‚РЅРѕРіРѕ С†РІРµС‚Р° в•ђв•ђв•ђ */
  useEffect(() => {
    document.body.dataset.accent = settings.accentColor
  }, [settings.accentColor])

  /* в•ђв•ђв•ђ РЎРёРЅС…СЂРѕРЅРёР·Р°С†РёСЏ Р°РІС‚РѕРїР»РµСЏ СЃ РЅР°СЃС‚СЂРѕР№РєР°РјРё в•ђв•ђв•ђ */
  useEffect(() => {
    setAutoplayActive(settings.autoplay)
  }, [settings.autoplay])

  /* в•ђв•ђв•ђ Toggle-С„СѓРЅРєС†РёРё в•ђв•ђв•ђ */
  const toggleCheatsheet = useCallback(() => setCheatsheetOpen(o => !o), [])
  const closeCheatsheet = useCallback(() => setCheatsheetOpen(false), [])

  const toggleSettings = useCallback(() => setSettingsOpen(o => !o), [])
  const closeSettings = useCallback(() => setSettingsOpen(false), [])

  const toggleShuffle = useCallback(() => setShuffleActive(s => !s), [])

  const toggleAutoplay = useCallback(() => {
    setAutoplayActive(prev => {
      const next = !prev
      settings.update('autoplay', next)
      return next
    })
  }, [settings])

  /* в•ђв•ђв•ђ Р РµРЅРґРµСЂ в•ђв•ђв•ђ */
  return (
    <div style={{ minHeight: '100vh', display: 'flex', flexDirection: 'column', background: 'var(--c-bg-soft)' }}>

      <Navbar
        onOpenCheatsheet={toggleCheatsheet}
        onOpenSettings={toggleSettings}
        onShuffle={toggleShuffle}
        shuffleActive={shuffleActive}
        onToggleAutoplay={toggleAutoplay}
        autoplayActive={autoplayActive}
      />

      <main style={{ flex: 1 }}>
        <Outlet context={{
          toggleCheatsheet,
          closeCheatsheet,
          toggleSettings,
          closeSettings,
          shuffleActive,
          toggleShuffle,
          autoplayActive,
          toggleAutoplay,
        }} />
      </main>

      <HotkeyCheatsheet isOpen={cheatsheetOpen} onClose={closeCheatsheet} />
      <SettingsPanel isOpen={settingsOpen} onClose={closeSettings} />

    </div>
  )
}

function shuffleArray<T>(arr: T[]): T[] {
  const result = [...arr]
  for (let i = result.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1))
    ;[result[i], result[j]] = [result[j], result[i]]
  }
  return result
}

  } = useOutletContext<OutletCtx>()

  const settings = useSettings()
  const stageRef = useRef<AnimalStageHandle>(null)
  const isPlayingRef = useRef(false)

  const [activeCategory, setActiveCategory] = useState('all')
  const [currentIndex, setCurrentIndex] = useState(0)
  const [shuffleSeed, setShuffleSeed] = useState(0)

  const baseFiltered = useMemo(() => {
    if (activeCategory === 'all') return animals
    return animals.filter(a => a.categories.includes(activeCategory))
  }, [activeCategory])

  const filtered = useMemo(() => {
    if (!shuffleActive) return baseFiltered
    return shuffleArray(baseFiltered)
  }, [baseFiltered, shuffleActive, shuffleSeed])

  const animal = filtered[currentIndex] ?? filtered[0] ?? animals[0]

  useEffect(() => {
    setCurrentIndex(0)
  }, [activeCategory, shuffleActive, shuffleSeed])

  const goNext = useCallback(() => {
    setCurrentIndex(i => {
      if (i >= filtered.length - 1) {
        return settings.loopNavigation ? 0 : i
      }
      return i + 1
    })
  }, [filtered.length, settings.loopNavigation])

  const goPrev = useCallback(() => {
    setCurrentIndex(i => {
      if (i <= 0) {
        return settings.loopNavigation ? filtered.length - 1 : 0
      }
      return i - 1
    })
  }, [filtered.length, settings.loopNavigation])

  const playSound = useCallback(() => {
    stageRef.current?.playSound()
  }, [])

  /* РђРІС‚РѕРїР»РµР№ вЂ” Р·РІСѓРє */
  useEffect(() => {
    if (!autoplayActive || !settings.autoplay) return
    const timer = setTimeout(() => {
      if (!isPlayingRef.current && stageRef.current) {
        isPlayingRef.current = true
        stageRef.current.playSound()
        setTimeout(() => { isPlayingRef.current = false }, 2000)
      }
    }, settings.autoplayDelay)
    return () => clearTimeout(timer)
  }, [animal?.id, autoplayActive, settings.autoplay, settings.autoplayDelay])

  /* РђРІС‚Рѕ-РїРµСЂРµС…РѕРґ Рє СЃР»РµРґСѓСЋС‰РµРјСѓ */
  useEffect(() => {
    if (!settings.autoNext || !autoplayActive) return
    if (filtered.length <= 1) return
    const timer = setInterval(goNext, settings.autoNextInterval)
    return () => clearInterval(timer)
  }, [settings.autoNext, settings.autoNextInterval, filtered.length, autoplayActive, goNext])

  /* РљР»Р°РІРёР°С‚СѓСЂР° вЂ” РїСЂР°РІРёР»СЊРЅС‹Рµ РёРјРµРЅР° РєРѕР»Р±СЌРєРѕРІ */
  useKeyboard({
    onNext: goNext,
    onPrev: goPrev,
    onPlay: playSound,
    onCheatsheet: toggleCheatsheet,
    onEscape: closeCheatsheet,
    onSettings: toggleSettings,
    onMute: () => settings.update('volume', settings.volume === 0 ? 80 : 0),
  })

  useSwipe({ onLeft: goNext, onRight: goPrev })

  if (!animal) return null

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: 'calc(100vh - 56px)' }}>
      <AnimalStage
        ref={stageRef}
        animal={animal}
        index={currentIndex}
        total={filtered.length}
        onPrev={goPrev}
        onNext={goNext}
        muted={settings.volume === 0}
        volume={settings.volume / 100}
        transitionType={settings.transitionType}
        playAllVariants={settings.playAllVariants}
      />
      <CategoryBar
        categories={categories}
        activeId={activeCategory}
        onChange={setActiveCategory}
        allCategoryIds={allCategoryIds}
        showSilent={settings.showSilent}
      />
    </div>
  )
}

  onPrev?: () => void

export function useKeyboard(actions: KeyboardActions) {
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      const tag = (e.target as HTMLElement)?.tagName
      if (tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT') return
      if (e.ctrlKey || e.metaKey || e.altKey) return

      const key = e.key
      const code = e.code

      switch (true) {
        // РќР°РІРёРіР°С†РёСЏ РІРїРµСЂС‘Рґ: в†’ в†“ D(en) Р’(ru)
        case key === 'ArrowRight' || key === 'ArrowDown' ||
             code === 'KeyD':
          e.preventDefault()
          actions.onNext?.()
          break

        // РќР°РІРёРіР°С†РёСЏ РЅР°Р·Р°Рґ: в†ђ в†‘ A(en) Р¤(ru)
        case key === 'ArrowLeft' || key === 'ArrowUp' ||
             code === 'KeyA':
          e.preventDefault()
          actions.onPrev?.()
          break

        // Р’РѕСЃРїСЂРѕРёР·РІРµРґРµРЅРёРµ
        case key === ' ' || key === 'Enter':
          e.preventDefault()
          actions.onPlay?.()
          break

        // Mute: M(en) Р¬(ru)
        case code === 'KeyM':
          e.preventDefault()
          actions.onMute?.()
          break

        // РќР°СЃС‚СЂРѕР№РєРё: S(en) Р«(ru)
        case code === 'KeyS':
          e.preventDefault()
          actions.onSettings?.()
          break

        // РџРѕР»РЅРѕСЌРєСЂР°РЅРЅС‹Р№: F(en) Рђ(ru)
        case code === 'KeyF':
          e.preventDefault()
          actions.onFullscreen?.()
          break

        // РЁРїР°СЂРіР°Р»РєР°: H(en) Р (ru) РёР»Рё ?
        case code === 'KeyH' || key === '?':
          e.preventDefault()
          actions.onCheatsheet?.()
          break

        case key === 'Home':
          e.preventDefault()
          actions.onFirst?.()
          break

        case key === 'End':
          e.preventDefault()
          actions.onLast?.()
          break

        case key === 'Escape':
          e.preventDefault()
          actions.onEscape?.()
          break

        // Р¦РёС„СЂС‹ 1-9 вЂ” РІС‹Р±РѕСЂ РєР°С‚РµРіРѕСЂРёРё
        case key >= '1' && key <= '9':
          e.preventDefault()
          actions.onCategory?.(parseInt(key) - 1)
          break
      }
    }

    window.addEventListener('keydown', handler)
    return () => window.removeEventListener('keydown', handler)
  }, [actions])
}

interface SwipeActions {
  onSwipeLeft?: () => void
  onSwipeRight?: () => void
}

const MIN_DISTANCE = 50

export function useSwipe(actions: SwipeActions) {
  const startX = useRef(0)
  const startY = useRef(0)

  const onTouchStart = useCallback((e: React.TouchEvent) => {
    startX.current = e.touches[0].clientX
    startY.current = e.touches[0].clientY
  }, [])

  const onTouchEnd = useCallback((e: React.TouchEvent) => {
    const dx = e.changedTouches[0].clientX - startX.current
    const dy = e.changedTouches[0].clientY - startY.current

    // РЎРІР°Р№Рї С‚РѕР»СЊРєРѕ РµСЃР»Рё РіРѕСЂРёР·РѕРЅС‚Р°Р»СЊРЅРѕРµ РґРІРёР¶РµРЅРёРµ Р±РѕР»СЊС€Рµ РІРµСЂС‚РёРєР°Р»СЊРЅРѕРіРѕ
    if (Math.abs(dx) > Math.abs(dy) && Math.abs(dx) > MIN_DISTANCE) {
      if (dx < 0) {
        actions.onSwipeLeft?.()  // РЎРІР°Р№Рї РІР»РµРІРѕ = СЃР»РµРґСѓСЋС‰РµРµ
      } else {
        actions.onSwipeRight?.() // РЎРІР°Р№Рї РІРїСЂР°РІРѕ = РїСЂРµРґС‹РґСѓС‰РµРµ
      }
    }
  }, [actions])

  return { onTouchStart, onTouchEnd }
}
